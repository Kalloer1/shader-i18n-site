import { onMounted } from 'vue'

onMounted(() => {
  // 监听标签页切换，懒加载内容

  const downloadTab = document.querySelector('label[for="sp-tab-download"]')

  // 介绍标签页
  const descContainer = document.getElementById('modrinth-desc')
  if (descContainer && descTab) {
    const projectId = descContainer.dataset.projectId
    let loaded = false
    descTab.addEventListener('click', () => {
      if (!loaded) { loaded = true; renderDescription(projectId, descContainer) }
    })
  }

  // 下载标签页（版本列表）
  const versionsContainer = document.getElementById('modrinth-versions')
  if (versionsContainer && downloadTab) {
    const projectId = versionsContainer.dataset.projectId
    let loaded = false
    // 下载标签默认选中，立即加载
    loaded = true
    renderVersions(projectId, versionsContainer)
  }
})
